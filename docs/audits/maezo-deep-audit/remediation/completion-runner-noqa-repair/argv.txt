import ast
import hashlib
import marshal
import subprocess
from pathlib import Path

path = Path('scripts/dev/run_engine_integration.py')
base = subprocess.run(
    ['git', 'show', '516902641c03208c2d2838b2b1d44749dd460cae:scripts/dev/run_engine_integration.py'],
    check=True,
    capture_output=True,
).stdout
current = path.read_bytes()
base_text = base.decode('utf-8')
current_text = current.decode('utf-8')
def ast_digest(text: str) -> str:
    return hashlib.sha256(ast.dump(ast.parse(text), annotate_fields=True, include_attributes=True).encode()).hexdigest()
def bytecode_digest(text: str) -> str:
    return hashlib.sha256(marshal.dumps(compile(text, str(path), 'exec'))).hexdigest()
print(f'base_source_sha256={hashlib.sha256(base).hexdigest()}')
print(f'current_source_sha256={hashlib.sha256(current).hexdigest()}')

print(f'current_lines={len(current_text.splitlines())}')
print(f'base_ast_sha256={ast_digest(base_text)}')
print(f'current_ast_sha256={ast_digest(current_text)}')
print(f'base_bytecode_sha256={bytecode_digest(base_text)}')
print(f'current_bytecode_sha256={bytecode_digest(current_text)}')
assert len(base_text.splitlines()) == len(current_text.splitlines()) == 2046
